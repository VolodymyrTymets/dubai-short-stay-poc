# Extra destructive command patterns for this repo, one JavaScript regex per line, case-insensitive.
# Read by the guard-destructive hook and blocked with rule C4.
# Core already blocks: rm -rf outside build dirs, git clean -fdx, DROP/TRUNCATE, schema resets,
# docker volume destruction, infra CLIs, publish/deploy/submit, non-local database hosts, curl | sh.

# Prisma (api/)
prisma migrate reset
prisma db push --force-reset
prisma db push --accept-data-loss

# api/package.json's own script names — the guard must catch these even though the literal
# script body ("npx prisma migrate reset") never appears in the invoking command.
\breset-db\b

# docker-compose.yml defines named volumes (pgdata, redisdata) for local Postgres/Redis state.
docker-compose\s+down\s+.*-v\b
docker\s+compose\s+down\s+.*--volumes\b
docker\s+volume\s+rm
